error_reporting(0); global $wpdb;
$in = $GLOBALS['pw_ops']; $kk = (string) $in['k']; $dry = !empty($in['dry']); $t0 = microtime(true);

$st = $b['art'][$kk] ?? null; if (!$st) return ['stop' => 'articolo assente'];
$pid = (int) $st['post']; $p = get_post($pid);
if (!$p || $p->post_status !== 'publish' || $p->post_type !== 'post' || strpos($p->post_title, $st['titolo']) !== 0) return ['stop' => 'controllo articolo fallito', 'title' => $p ? $p->post_title : null];

foreach ($st['ops'] as $i => $op) {
  if ($op[0] !== 're') return ['stop' => 'op sconosciuta ' . $i];
  $pat = '~' . $op[1] . '~s'; $n = preg_match_all($pat, $new, $mm, PREG_OFFSET_CAPTURE); if ($n !== 1) return ['stop' => 'op ' . $i . ': corrispondenze ' . var_export($n, true), 'k' => $kk];

}
if ($new === $cur) return ['stop' => 'nessuna modifica'];

$chk = $wpdb->get_var($wpdb->prepare("SELECT post_content FROM {$wpdb->posts} WHERE ID=%d", $pid));
update_post_meta($pid, $mk, $in['sha'] . '|' . current_time('mysql'));
return ['k' => $kk, 'post' => $pid, 'written' => $chk === $new, 'len' => [strlen($cur), strlen($new)], 'sec' => round(microtime(true) - $t0, 1)];
